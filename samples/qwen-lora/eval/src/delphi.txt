--- ref
program QuickSort;
{$APPTYPE CONSOLE}
uses
  System.SysUtils;
type
  TSort = class
  public
    class procedure Sort(var A: TArray<Integer>; Lo, Hi: Integer); static;
  end;

class procedure TSort.Sort(var A: TArray<Integer>; Lo, Hi: Integer);
var
  I, J, Pivot, T: Integer;
begin
  if Lo >= Hi then
    Exit;
  Pivot := A[Hi];
  I := Lo;
  for J := Lo to Hi - 1 do
    if A[J] <= Pivot then
    begin
      T := A[I];
      A[I] := A[J];
      A[J] := T;
      Inc(I);
    end;
  T := A[I];
  A[I] := A[Hi];
  A[Hi] := T;
  Sort(A, Lo, I - 1);
  Sort(A, I + 1, Hi);
end;

var
  A: TArray<Integer>;
  I: Integer;
begin
  A := [29, 3, 72, 15, 8, 41, 3, 56];
  TSort.Sort(A, 0, High(A));
  for I := 0 to High(A) do
  begin
    if I > 0 then
      Write(' ');
    Write(A[I]);
  end;
  Writeln;

  System.SysUtils;

function Factorise(N: Integer): string;

  P, E: Integer;
begin
  Result := '';
  P := 2;
  while N > 1 do
  begin
    E := 0;
    while N mod P = 0 do

      N := N div P;

    end;
    if E > 0 then
    begin
      if Result <> '' then
        Result := Result + ' * ';
      Result := Result + IntToStr(P);

        Result := Result + '^' + IntToStr(E);
    end;
    Inc(P);
  end;

--- prompt
Simulate a queue at a counter: customers arrive at minutes 0, 1, 2, 6, 7 and each needs 3 minutes of service; one clerk serves in arrival order. Use a TQueue<Integer> of arrival times and print for each customer "customer n waits w" and finally the average wait with two decimals.
--- ref
program Counter;
{$APPTYPE CONSOLE}
uses
  System.SysUtils, System.Generics.Collections;
const
  Arrivals: array[0..4] of Integer = (0, 1, 2, 6, 7);
var
  Q: TQueue<Integer>;
  Free, Arrive, Wait, N, Total: Integer;
  FS: TFormatSettings;
begin
  FS := TFormatSettings.Create;
  FS.DecimalSeparator := '.';
  Q := TQueue<Integer>.Create;
  try
    for var T in Arrivals do
      Q.Enqueue(T);
    Free := 0;
    N := 0;
    Total := 0;
    while Q.Count > 0 do
    begin
      Arrive := Q.Dequeue;
      Inc(N);
      if Free < Arrive then
        Free := Arrive;
      Wait := Free - Arrive;
      Inc(Total, Wait);
      Writeln('customer ', N, ' waits ', Wait);
      Free := Free + 3;
    end;
    Writeln('average ', FormatFloat('0.00', Total / N, FS));
  finally
    Q.Free;
  end;
end.

--- ref
program SmallSet;

  System.SysUtils;
type
  TSmallSet = record
    Bits: Cardinal;
    procedure Include(N: Integer);
    function Contains(N: Integer): Boolean;
    function Count: Integer;
  end;

procedure TSmallSet.Include(N: Integer);
begin
  Bits := Bits or (Cardinal(1) shl N);
end;

function TSmallSet.Contains(N: Integer): Boolean;
begin
  Result := (Bits and (Cardinal(1) shl N)) <> 0;
end;

function TSmallSet.Count: Integer;
begin
  Result := 0;
  for var I := 0 to 31 do
    if Contains(I) then
      Inc(Result);
end;

const
  Added: array[0..4] of Integer = (1, 3, 5, 7, 3);
var
  S: TSmallSet;
begin
  S.Bits := 0;
  for var N in Added do
    S.Include(N);
  Writeln('contains 3: ', BoolToStr(S.Contains(3), True));
  Writeln('contains 4: ', BoolToStr(S.Contains(4), True));
  Writeln('count ', S.Count);

--- ref
program Polygon;
{$APPTYPE CONSOLE}
uses
  System.SysUtils;
const
  X: array[0..4] of Double = (0, 4, 4, 2, 0);
  Y: array[0..4] of Double = (0, 0, 3, 5, 3);
var
  Area, Perim: Double;
  J: Integer;
  FS: TFormatSettings;
begin
  FS := TFormatSettings.Create;
  FS.DecimalSeparator := '.';
  Area := 0;
  Perim := 0;
  for var I := 0 to 4 do
  begin
    J := (I + 1) mod 5;
    Area := Area + X[I] * Y[J] - X[J] * Y[I];
    Perim := Perim + Sqrt(Sqr(X[J] - X[I]) + Sqr(Y[J] - Y[I]));
  end;
  Writeln('area ', FormatFloat('0.0', Abs(Area) / 2, FS));
  Writeln('perimeter ', FormatFloat('0.00', Perim, FS));
end.

Write a class TMemo with a TStringList history supporting Write(text), Undo and Text (the lines joined by " | "). Write "a", "b", "c", undo once, write "d", undo twice, and print the text after each step as "step: text".
--- ref
program UndoMemo;
{$APPTYPE CONSOLE}
uses
  System.SysUtils, System.Classes;
type
  TMemo = class
  private
    FLines: TStringList;
  public
    constructor Create;
    destructor Destroy; override;
    procedure WriteLine(const S: string);
    procedure Undo;
    function Text: string;
  end;

constructor TMemo.Create;
begin
  inherited Create;
  FLines := TStringList.Create;
end;

destructor TMemo.Destroy;
begin
  FLines.Free;
  inherited;
end;

procedure TMemo.WriteLine(const S: string);
begin
  FLines.Add(S);
end;

procedure TMemo.Undo;
begin
  if FLines.Count > 0 then
    FLines.Delete(FLines.Count - 1);
end;

function TMemo.Text: string;
begin
  Result := '';
  for var I := 0 to FLines.Count - 1 do
  begin
    if I > 0 then
      Result := Result + ' | ';
    Result := Result + FLines[I];
  end;
end;

var
  M: TMemo;
begin
  M := TMemo.Create;
  try
    M.WriteLine('a');
    M.WriteLine('b');
    M.WriteLine('c');
    Writeln('write: ', M.Text);
    M.Undo;
    Writeln('undo: ', M.Text);
    M.WriteLine('d');
    Writeln('write: ', M.Text);
    M.Undo;
    M.Undo;
    Writeln('undo2: ', M.Text);
  finally
    M.Free;
  end;

--- ref
program Prefix;
{$APPTYPE CONSOLE}
uses
  System.SysUtils;

function CommonPrefix(const A: array of string): string;
begin
  Result := A[0];
  for var I := 1 to High(A) do
    while not A[I].StartsWith(Result) do
      Result := Result.Substring(0, Result.Length - 1);
end;

procedure Show(const A: array of string);
var
  P: string;
begin
  P := CommonPrefix(A);
  if P = '' then
    Writeln('(none)')
  else
    Writeln(P);
end;

begin
  Show(['interstellar', 'internet', 'interval', 'internal']);
  Show(['dog', 'racecar', 'car']);
end.

Implement a minimal matrix-free "spreadsheet": cells A1=5, A2=7, A3=A1+A2, A4=A3*2 stored as formulas in a TDictionary<string, string>, evaluated recursively (a formula is a number, or two cell names joined by + or *). Print each cell as "A3 = 12" in order A1 to A4.
--- ref
program Sheet;
{$APPTYPE CONSOLE}
uses
  System.SysUtils, System.Generics.Collections;
var
  Cells: TDictionary<string, string>;

function Eval(const Name: string): Integer;
var
  F: string;
  P: Integer;
begin
  F := Cells[Name];
  P := F.IndexOf('+');
  if P >= 0 then
    Exit(Eval(F.Substring(0, P)) + Eval(F.Substring(P + 1)));
  P := F.IndexOf('*');
  if P >= 0 then
  begin
    if Cells.ContainsKey(F.Substring(P + 1)) then
      Exit(Eval(F.Substring(0, P)) * Eval(F.Substring(P + 1)))
    else
      Exit(Eval(F.Substring(0, P)) * StrToInt(F.Substring(P + 1)));
  end;
  Result := StrToInt(F);
end;

begin
  Cells := TDictionary<string, string>.Create;
  try
    Cells.Add('A1', '5');
    Cells.Add('A2', '7');
    Cells.Add('A3', 'A1+A2');
    Cells.Add('A4', 'A3*2');
    for var N in ['A1', 'A2', 'A3', 'A4'] do
      Writeln(N, ' = ', Eval(N));
  finally
    Cells.Free;
  end;
end.

function NextTerm(const S: string): string;
var
  I, N: Integer;
begin
  Result := '';
  I := 1;
  while I <= S.Length do
  begin
    N := 1;
    while (I + N <= S.Length) and (S[I + N] = S[I]) do
      Inc(N);
    Result := Result + IntToStr(N) + S[I];
    Inc(I, N);
  end;
end;

  for var I := 1 to 10 do
