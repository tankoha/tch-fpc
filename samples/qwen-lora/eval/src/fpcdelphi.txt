procedure Permute(const Prefix, Rest: string; AOut: TStrings);

  I: Integer;

  if Rest = '' then

class procedure TSorter.Sort<T>(var A: array of T; Compare: TCompare<T>);
var
  I, J: Integer;
  Tmp: T;
begin
  for I := High(A) downto 1 do
    for J := 0 to I - 1 do
      if Compare(A[J], A[J + 1]) > 0 then
      begin
        Tmp := A[J];
        A[J] := A[J + 1];
        A[J + 1] := Tmp;
      end;
end;

end;

program Brackets;
{$mode delphi}
uses
  SysUtils, Generics.Collections;

function Balanced(const S: string): Boolean;
var
  St: TStack<Char>;
  C: Char;
begin
  St := TStack<Char>.Create;
  try
    for C in S do
      case C of
        '(', '[', '{': St.Push(C);
        ')', ']', '}':
          begin
            if St.Count = 0 then
              Exit(False);
            if ((C = ')') and (St.Peek <> '(')) or ((C = ']') and (St.Peek <> '[')) or
               ((C = '}') and (St.Peek <> '{')) then
              Exit(False);
            St.Pop;
          end;
      end;
    Result := St.Count = 0;
  finally
    St.Free;
  end;
end;

procedure Check(const S: string);
begin
  if Balanced(S) then
    WriteLn(S, ' balanced')
  else
    WriteLn(S, ' unbalanced');
end;

  Check('([)]');
  Check('((x)');

program Fractions;
{$mode delphi}
uses
  SysUtils;
type
  TFraction = record
    Num, Den: Integer;
    class function Make(N, D: Integer): TFraction; static;
    class operator Add(const A, B: TFraction): TFraction;
    class operator Multiply(const A, B: TFraction): TFraction;
    function ToString: string;
  end;

function Gcd(A, B: Integer): Integer;
begin
  A := Abs(A);
  B := Abs(B);
  while B <> 0 do
  begin
    Result := A mod B;
    A := B;
    B := Result;
  end;
  Result := A;
end;

class function TFraction.Make(N, D: Integer): TFraction;
var
  G: Integer;
begin
  if D < 0 then
  begin
    N := -N;
    D := -D;
  end;
  G := Gcd(N, D);
  if G = 0 then
    G := 1;
  Result.Num := N div G;
  Result.Den := D div G;
end;

class operator TFraction.Add(const A, B: TFraction): TFraction;
begin
  Result := Make(A.Num * B.Den + B.Num * A.Den, A.Den * B.Den);
end;

class operator TFraction.Multiply(const A, B: TFraction): TFraction;
begin
  Result := Make(A.Num * B.Num, A.Den * B.Den);
end;

function TFraction.ToString: string;
begin
  Result := IntToStr(Num) + '/' + IntToStr(Den);
end;

program RunLength;
{$mode delphi}
uses
  SysUtils;

function Encode(const S: string): string;
var
  I, N: Integer;
begin
  Result := '';
  I := 1;
  while I <= Length(S) do
  begin
    N := 1;
    while (I + N <= Length(S)) and (S[I + N] = S[I]) do
      Inc(N);
    Result := Result + IntToStr(N) + S[I];
    Inc(I, N);
  end;
end;

function Decode(const S: string): string;
var
  I, N: Integer;
begin
  Result := '';
  N := 0;
  for I := 1 to Length(S) do
    if S[I] in ['0'..'9'] then
      N := N * 10 + Ord(S[I]) - Ord('0')
    else
    begin
      Result := Result + StringOfChar(S[I], N);
      N := 0;
    end;
end;

var
  E: string;
begin
  E := Encode('aaabccddddde');
  WriteLn(E);
  WriteLn(Decode(E));
end.

      Best[I, C] := Best[I - 1, C];
      if (W[I] <= C) and (Best[I - 1, C - W[I]] + V[I] > Best[I, C]) then

    end;
