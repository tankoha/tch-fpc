program Sieve;
{$mode objfpc}{$H+}
const
  Limit = 59;
var
  Composite: array[2..Limit] of Boolean;
  I, J, N: Integer;
begin
  for I := 2 to Limit do
    Composite[I] := False;
  for I := 2 to Limit do
    if not Composite[I] then
    begin
      J := I * I;
      while J <= Limit do
      begin
        Composite[J] := True;
        Inc(J, I);
      end;
    end;
  N := 0;
  for I := 2 to Limit do
    if not Composite[I] then
    begin
      WriteLn(I);
      Inc(N);
    end;
  WriteLn('count: ', N);
end.

program InsertSort;
{$mode objfpc}{$H+}

  A: array of Integer;
  I, J, Key: Integer;
begin
  A := [42, 7, 19, 3, 25, 11, 8];
  for I := 1 to High(A) do
  begin
    Key := A[I];
    J := I - 1;
    while (J >= 0) and (A[J] > Key) do
    begin
      A[J + 1] := A[J];
      Dec(J);
    end;
    A[J + 1] := Key;
  end;
  for I := 0 to High(A) do
  begin
    if I > 0 then
      Write(',');
    Write(A[I]);
  end;
  WriteLn;
end.

type
  TIntStack = class
  private
    FItems: array of Integer;
    FCount: Integer;
  public
    procedure Push(AValue: Integer);
    function Pop: Integer;
    function IsEmpty: Boolean;
  end;

procedure TIntStack.Push(AValue: Integer);
begin
  if FCount = Length(FItems) then
    SetLength(FItems, FCount * 2 + 4);
  FItems[FCount] := AValue;
  Inc(FCount);
end;

function TIntStack.Pop: Integer;
begin
  Dec(FCount);
  Result := FItems[FCount];
end;

function TIntStack.IsEmpty: Boolean;
begin
  Result := FCount = 0;
end;

  S: TIntStack;
  I: Integer;
begin
  S := TIntStack.Create;
  try
    for I := 1 to 5 do
      S.Push(I);
    while not S.IsEmpty do
      WriteLn(S.Pop);
  finally
    S.Free;
  end;

function ToBinary(N: Integer): string;
begin
  if N = 0 then
    Exit('0');
  Result := '';
  while N > 0 do
  begin
    Result := Chr(Ord('0') + N mod 2) + Result;
    N := N div 2;
  end;
end;

function FromBinary(const S: string): Integer;
var
  I: Integer;
begin
  Result := 0;
  for I := 1 to Length(S) do
    Result := Result * 2 + Ord(S[I]) - Ord('0');

  WriteLn(FromBinary('101101'));
end.

function Find(AValue: Integer): Integer;

  Lo, Hi, Mid: Integer;

  Lo := 0;
  Hi := High(Data);

      Lo := Mid + 1

function Neighbours(const G: TGrid; R, C: Integer): Integer;

  DR, DC: Integer;
begin
  Result := 0;
  for DR := -1 to 1 do
    for DC := -1 to 1 do
      if ((DR <> 0) or (DC <> 0)) and (R + DR >= 0) and (R + DR < N) and
         (C + DC >= 0) and (C + DC < N) and G[R + DR, C + DC] then
        Inc(Result);

  Next: TGrid;
  R, C, K: Integer;

  Classes, SysUtils;

function Eval(const AExpr: string): Integer;
var
  Parts: TStringList;
  Stack: array of Integer;
  N, I, A, B: Integer;
  T: string;
begin
  Parts := TStringList.Create;
  try
    Parts.Delimiter := ' ';
    Parts.StrictDelimiter := True;
    Parts.DelimitedText := AExpr;
    Stack := nil;
    SetLength(Stack, Parts.Count);
    N := 0;
    for I := 0 to Parts.Count - 1 do
    begin
      T := Parts[I];
      if (T = '+') or (T = '-') or (T = '*') then
      begin
        B := Stack[N - 1];
        A := Stack[N - 2];
        Dec(N, 2);
        case T[1] of
          '+': Stack[N] := A + B;
          '-': Stack[N] := A - B;
          '*': Stack[N] := A * B;
        end;
      end
      else
        Stack[N] := StrToInt(T);
      Inc(N);
    end;
    Result := Stack[0];
  finally
    Parts.Free;
  end;
end;

program RingQueue;
{$mode objfpc}{$H+}
uses
  SysUtils;
type
  TRingQueue = class
  private
    FItems: array[0..3] of string;
    FHead, FCount: Integer;
  public
    procedure Enqueue(const S: string);
    function Dequeue: string;
    property Count: Integer read FCount;
  end;

procedure TRingQueue.Enqueue(const S: string);
begin
  if FCount = Length(FItems) then
    raise Exception.Create('queue full, cannot add ' + S);
  FItems[(FHead + FCount) mod Length(FItems)] := S;
  Inc(FCount);
end;

function TRingQueue.Dequeue: string;
begin
  if FCount = 0 then
    raise Exception.Create('queue empty');
  Result := FItems[FHead];
  FHead := (FHead + 1) mod Length(FItems);
  Dec(FCount);
end;

  Q: TRingQueue;

  Q := TRingQueue.Create;
  try
    Q.Enqueue('a');
    Q.Enqueue('b');
    Q.Enqueue('c');
    WriteLn(Q.Dequeue);
    WriteLn(Q.Dequeue);
    Q.Enqueue('d');
    Q.Enqueue('e');
    Q.Enqueue('f');
    try
      Q.Enqueue('g');
    except
      on E: Exception do
        WriteLn(E.Message);
    end;
    while Q.Count > 0 do
      WriteLn(Q.Dequeue);
  finally
    Q.Free;
  end;
